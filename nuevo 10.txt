    <script>
        // Horario actualizado con los datos del usuario
        const schedule = {
            "Lunes": ["Geografía", "Id. Español", "Biología", "Ciencias de la comp."],
            "Martes": ["Matemática", "Portugués", "Cs. Físico-Química", "Ed. Física y Rec."],
            "Miércoles": ["Id. Español", "Cs. Físico-Química", "Historia", "Ed. Musical", "J2"],
            "Jueves": ["Geografía", "Historia", "Matemática", "Com. Visual", "Yoga", "Ed. Física y Rec."],
            "Viernes": ["Ciencias de la comp.", "Matemática", "Biología", "Téc. Estudio", "Inglés", "J2"]
        };

        let tasks = JSON.parse(localStorage.getItem('tasks_7mo')) || [];

        window.onload = () => {
            renderSchedule();
            renderTasks();
            checkReminders();
        };

        function renderSchedule() {
            const container = document.getElementById('schedule-list');
            container.innerHTML = '';
            
            Object.keys(schedule).forEach(day => {
                const div = document.createElement('div');
                div.className = "border-l-2 border-indigo-200 pl-4 py-1";
                div.innerHTML = `
                    <h4 class="font-bold text-indigo-700 mb-1">${day}</h4>
                    <div class="flex flex-wrap gap-1">
                        ${schedule[day].map(sub => `<span class="text-xs bg-indigo-50 text-indigo-600 px-2 py-1 rounded-md border border-indigo-100">${sub}</span>`).join('')}
                    </div>
                `;
                container.appendChild(div);
            });
        }

        function renderTasks() {
            const container = document.getElementById('tasks-container');
            container.innerHTML = '';
            
            if (tasks.length === 0) {
                container.innerHTML = '<div class="text-center py-10"><p class="text-slate-400 italic">No tienes tareas pendientes. ¡Libre!</p></div>';
                return;
            }

            tasks.sort((a, b) => new Date(a.date) - new Date(b.date)).forEach(task => {
                const isUrgent = isTomorrow(task.date);
                const div = document.createElement('div');
                div.className = `group flex items-center justify-between p-4 rounded-xl border transition-all ${task.completed ? 'bg-slate-50 border-slate-200 opacity-60' : (isUrgent ? 'bg-red-50 border-red-200 shadow-sm' : 'bg-white border-slate-100 shadow-sm')}`;
                
                div.innerHTML = `
                    <div class="flex items-center gap-4">
                        <input type="checkbox" ${task.completed ? 'checked' : ''} onchange="toggleTask(${task.id})" class="w-6 h-6 rounded-full border-slate-300 text-indigo-600 focus:ring-indigo-500">
                        <div>
                            <p class="font-bold ${task.completed ? 'line-through text-slate-400' : 'text-slate-800'}">${task.name}</p>
                            <p class="text-xs ${isUrgent && !task.completed ? 'text-red-500 font-bold' : 'text-slate-500'} italic">
                                ${isUrgent && !task.completed ? '⚠️ ENTREGA MAÑANA' : 'Fecha: ' + formatDate(task.date)}
                            </p>
                        </div>
                    </div>
                    <button onclick="deleteTask(${task.id})" class="text-slate-300 hover:text-red-500 transition-colors">
                        <svg class="w-5 h-5" fill="none" stroke="currentColor" viewBox="0 0 24 24"><path stroke-linecap="round" stroke-linejoin="round" stroke-width="2" d="M19 7l-.867 12.142A2 2 0 0116.138 21H7.862a2 2 0 01-1.995-1.858L5 7m5 4v6m4-6v6m1-10V4a1 1 0 00-1-1h-4a1 1 0 00-1 1v3M4 7h16"></path></svg>
                    </button>
                `;
                container.appendChild(div);
            });
            localStorage.setItem('tasks_7mo', JSON.stringify(tasks));
            checkReminders();
        }

        function addTask() {
            const name = document.getElementById('task-name').value;
            const date = document.getElementById('task-date').value;

            if (!name || !date) {
                showModal("Faltan datos", "Escribe el nombre de la tarea y cuándo hay que entregarla.");
                return;
            }

            tasks.push({ id: Date.now(), name, date, completed: false });
            document.getElementById('task-name').value = '';
            document.getElementById('task-date').value = '';
            renderTasks();
        }

        function toggleTask(id) {
            tasks = tasks.map(t => t.id === id ? { ...t, completed: !t.completed } : t);
            renderTasks();
        }

        function deleteTask(id) {
            tasks = tasks.filter(t => t.id !== id);
            renderTasks();
        }

        function isTomorrow(dateStr) {
            const today = new Date();
            const tomorrow = new Date(today);
            tomorrow.setDate(tomorrow.getDate() + 1);
            return dateStr === tomorrow.toISOString().split('T')[0];
        }

        function formatDate(dateStr) {
            const [year, month, day] = dateStr.split('-');
            return `${day}/${month}/${year}`;
        }

        function checkReminders() {
            const today = new Date();
            const tomorrow = new Date(today);
            tomorrow.setDate(tomorrow.getDate() + 1);
            
            const daysMap = ["Domingo", "Lunes", "Martes", "Miércoles", "Jueves", "Viernes", "Sábado"];
            const tomorrowDayName = daysMap[tomorrow.getDay()];
            const tomorrowISO = tomorrow.toISOString().split('T')[0];

            // Materias
            const subjectsElement = document.getElementById('next-day-subjects');
            if (schedule[tomorrowDayName]) {
                subjectsElement.innerText = `Mañana tienes: ${schedule[tomorrowDayName].join(', ')}`;
            } else {
                subjectsElement.innerText = "¡No hay clases mañana!";
            }

            // Tareas
            const urgentElement = document.getElementById('urgent-tasks');
            const urgent = tasks.filter(t => t.date === tomorrowISO && !t.completed);
            if (urgent.length > 0) {
                urgentElement.innerHTML = `<span class="text-red-600">¡Tenés que terminar: ${urgent.map(t => t.name).join(', ')}!</span>`;
            } else {
                urgentElement.innerText = "Todo al día por ahora.";
            }
        }

        function handleCredentialResponse(response) {
            try {
                const base64Url = response.credential.split('.')[1];
                const base64 = base64Url.replace(/-/g, '+').replace(/_/g, '/');
                const jsonPayload = decodeURIComponent(window.atob(base64).split('').map(c => '%' + ('00' + c.charCodeAt(0).toString(16)).slice(-2)).join(''));
                const payload = JSON.parse(jsonPayload);
                document.getElementById('user-info').innerText = `Estudiante: ${payload.name}`;
                document.getElementById('user-info').classList.remove('hidden');
                showModal("¡Hola!", `Agenda vinculada con la cuenta de ${payload.given_name}.`);
            } catch (e) { console.error(e); }
        }

        function showModal(title, msg) {
            document.getElementById('modal-title').innerText = title;
            document.getElementById('modal-message').innerText = msg;
            document.getElementById('custom-modal').classList.remove('hidden');
        }

        function closeModal() {
            document.getElementById('custom-modal').classList.add('hidden');
        }
    </script>
